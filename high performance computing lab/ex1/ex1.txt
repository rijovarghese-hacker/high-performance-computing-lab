#include <stdio.h>
#include <time.h>
#include <stdlib.h>

double time_diff(struct timespec s, struct timespec e) {
    return (e.tv_sec - s.tv_sec) * 1000 +
           (e.tv_nsec - s.tv_nsec) / 1000;
}

int main() {
    int r, c, i, j;

    printf("Enter rows and columns: ");
    scanf("%d %d", &r, &c);

    int *A   = (int *)malloc(r * c * sizeof(int));
    int *B   = (int *)malloc(r * c * sizeof(int));
    int *Add = (int *)malloc(r * c * sizeof(int));
    int *Sub = (int *)malloc(r * c * sizeof(int));

    srand(time(NULL));
    for (i = 0; i < r; i++) {
        for (j = 0; j < c; j++) {
            A[i*c + j] = rand() % 10;
            B[i*c + j] = rand() % 10;
        }
    }

    struct timespec start, end;
    clock_gettime(CLOCK_MONOTONIC, &start);

    for (i = 0; i < r; i++)
        for (j = 0; j < c; j++)
            Add[i*c + j] = A[i*c + j] + B[i*c + j];

    for (i = 0; i < r; i++)
        for (j = 0; j < c; j++)
            Sub[i*c + j] = A[i*c + j] - B[i*c + j];

    clock_gettime(CLOCK_MONOTONIC, &end);

    printf("\nserial exec time: %lf \n",
           time_diff(start, end));

    free(A);
    free(B);
    free(Add);
    free(Sub);

    return 0;
}

float time_diff(struct timespec s, struct timespec e) {
    return (e.tv_sec - s.tv_sec) * 1000.0 + (e.tv_nsec - s.tv_nsec) / 1000.0;
}

int main() {
    int r, c, i;
    printf("Enter rows and columns: ");
    scanf("%d %d", &r, &c);

    int *A   = (int*) malloc(r * c * sizeof(int));
    int *B   = (int*) malloc(r * c * sizeof(int));
    int *Res = (int*) malloc(r * c * sizeof(int));

    srand(time(NULL));
    for (i = 0; i < r * c; i++) {
        A[i] = rand() % 10;
        B[i] = rand() % 10;
    }

    int fd[2];
    pipe(fd);

    pid_t pid1 = fork();
    if (pid1 == 0) {
        close(fd[0]);
        struct timespec cs, ce;
        clock_gettime(CLOCK_MONOTONIC, &cs);

        for (i = 0; i < r * c; i++) Res[i] = A[i] + B[i];

        clock_gettime(CLOCK_MONOTONIC, &ce);
        double c1_time = time_diff(cs, ce);
        write(fd[1], &c1_time, sizeof(c1_time));
        printf("child 1 (add) exec time: %f ms\n", c1_time);
        close(fd[1]);
        exit(0);
    }

    pid_t pid2 = fork();
    if (pid2 == 0) {
        close(fd[0]);
        struct timespec cs2, ce2;
        clock_gettime(CLOCK_MONOTONIC, &cs2);

        for (i = 0; i < r * c; i++) Res[i] = A[i] - B[i];

        clock_gettime(CLOCK_MONOTONIC, &ce2);
        double c2_time = time_diff(cs2, ce2);
        write(fd[1], &c2_time, sizeof(c2_time));
        printf("child 2 (sub) exec time: %f ms\n", c2_time);
        close(fd[1]);
        exit(0);
    }
    close(fd[1]);

    double c1_time, c2_time;

    read(fd[0], &c1_time, sizeof(c1_time));
    read(fd[0], &c2_time, sizeof(c2_time));

    wait(NULL);
    wait(NULL);

    // Calculate Parallel Time (The time the longest process took)
    double max_time = (c1_time > c2_time) ? c1_time : c2_time;

    printf("\ntotal parallel execution time : %f ms\n", max_time);

    close(fd[0]);
    free(A); free(B); free(Res);
    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/ipc.h>
#include <sys/shm.h>
#include <sys/wait.h>
#include <time.h>
int main() {
    int r1, c1, r2, c2, i, j, k;
    struct timespec start, end;

    srand(time(NULL));

    printf("Enter size of matrix A (rows cols): ");
    scanf("%d %d", &r1, &c1);
    printf("Enter size of matrix B (rows cols): ");
    scanf("%d %d", &r2, &c2);

    int *A = (int *)malloc(r1 * c1 * sizeof(int));
    int *B = (int *)malloc(r2 * c2 * sizeof(int));


    for (i = 0; i < r1 * c1; i++) A[i] = rand() % 10;
    for (i = 0; i < r2 * c2; i++) B[i] = rand() % 10;
    size_t size_C = r1 * c2 * sizeof(int);
    size_t size_Times = r1 * sizeof(double);

    int shmid_C = shmget(IPC_PRIVATE, size_C, IPC_CREAT | 0666);
    int shmid_T = shmget(IPC_PRIVATE, size_Times, IPC_CREAT | 0666);

    int *c = (int *)shmat(shmid_C, NULL, 0);
    double *times = (double *)shmat(shmid_T, NULL, 0);

    for (i = 0; i < r1; i++) {
        pid_t pid = fork();
        if (pid == 0) {
            clock_gettime(CLOCK_MONOTONIC, &start);
            for (j = 0; j < c2; j++) {
                int sum = 0;
                for (k = 0; k < c1; k++) {
                    sum += A[i * c1 + k] * B[k * c2 + j];
                }
                c[i * c2 + j] = sum;
            }

            clock_gettime(CLOCK_MONOTONIC, &end); // End timer
            double time_taken = (end.tv_sec - start.tv_sec)*1000+
                                (end.tv_nsec - start.tv_nsec) / 1000;
 times[i] = time_taken; // Store time in shared memory

            shmdt(c);
            shmdt(times);
            exit(0);
        }
    }

    for (i = 0; i < r1; i++) wait(NULL);

    // 4. Find Maximum Execution Time
    double max_time = 0;
    for (i = 0; i < r1; i++) {
        if (times[i] > max_time) max_time = times[i];
    }

    printf("\nMaximum Child Execution Time: %2f ms\n", max_time);
    // Cleanup
    free(A);
    free(B);
    shmdt(c);
    shmdt(times);
    shmctl(shmid_C, IPC_RMID, NULL);
    shmctl(shmid_T, IPC_RMID, NULL);
return 0;
}
