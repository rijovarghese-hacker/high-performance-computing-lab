#include <stdio.h>
#include <stdlib.h>
#include <omp.h>

int main()
{
    int n;
    printf("Enter the number of result: ");
    if (scanf("%d", &n) != 1) return 1;

    double *a = (double*)malloc(n * sizeof(double));
    double *b = (double*)malloc(n * sizeof(double));
    double *c = (double*)malloc(n * sizeof(double));
    int i;

    // Initialize arrays
    for (i = 0; i < n; i++) {
        a[i] = i;
        b[i] = i + 1;
    }

    FILE *fp = fopen("out.txt", "w");
    if (fp == NULL) {
        printf("Error opening file!\n");
        free(a); free(b); free(c);
        return 1;
    }

    // Start Total Execution Timer
    double total_start_time = omp_get_wtime();

    #pragma omp parallel
    {
        int tid = omp_get_thread_num();
        double thread_start = omp_get_wtime();

        #pragma omp for
        for (i = 0; i < n; i++) {
            c[i] = a[i] + b[i]
        }

        double thread_end = omp_get_wtime();
        printf("Thread %d time taken: %f seconds\n", tid, thread_end - thread_start);
    }

    // End Total Execution Timer
    double total_end_time = omp_get_wtime();

    // Print the final total time
    printf("\n--- Results ---\n");
    printf("Total parallel execution time: %f seconds\n", total_end_time - total_start_time);

    // Write to file
    for (i = 0; i < n; i++) {
        fprintf(fp, "Result: C[%d] : %.1f\n", i, c[i]);
    }

    // Print first 10 for verification
    int limit = (n < 10) ? n : 10;
    for(i = 0; i < limit; i++) {
        printf("Result: C[%d] : %.1f\n", i, c[i]);
    }

    fclose(fp);
    printf("Results successfully written to out.txt\n");

    free(a);
    free(b);
    free(c);

    return 0;
}
