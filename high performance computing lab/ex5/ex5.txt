#include <stdio.h>
#include <pthread.h>
#include <time.h>

#define SIZE 10

int rows, cols, colsB;
int Matrix1[SIZE][SIZE], Matrix2[SIZE][SIZE]
int aMatrix[SIZE][SIZE], mMatrix[SIZE][SIZE]

// Global variables to store the duration
double add_exec_time, mul_exec_time;

void* addMatrix(void* arg) {
    struct timespec start, end;
    printf("\nAddthread running...");
    // Start timing ONLY the computation
    clock_gettime(CLOCK_MONOTONIC, &start);
    int li,lj;
    for(li=0; li<rows; li++) {
        for(lj=0; lj<cols; lj++) {
            aMatrix[li][lj] = Matrix1[li][lj] + Matrix2[li][lj]
        }
    }
    printf("\nAddthread finished the execution...");
    clock_gettime(CLOCK_MONOTONIC, &end);
    // Calculate elapsed time in seconds
    add_exec_time = (end.tv_sec - start.tv_sec) + (end.tv_nsec - start.tv_nsec) / 1e9;
    return NULL;
}

void* mulMatrix(void* arg) {
    struct timespec start, end;
    printf("mulThread running....");
    // Start timing ONLY the computation
    clock_gettime(CLOCK_MONOTONIC, &start);
    int li,lj,lk;
    for(li=0; li<rows; li++) {
        for(lj=0; lj<colsB; lj++) {
            mMatrix[li][lj] = 0;
            for(lk=0; lk<cols; lk++) {
                mMatrix[li][lj] += Matrix1[li][lk] * Matrix2[lk][lj]
            }
        }
    }
    printf("mulThread finished its task");
    clock_gettime(CLOCK_MONOTONIC, &end);
    mul_exec_time = (end.tv_sec - start.tv_sec) + (end.tv_nsec - start.tv_nsec) / 1e9;
    return NULL;
}

int main() {
    pthread_t addThread, mulThread;
    int mi, mj;

    // 1. Get Input from User
    printf("Enter Rows and Cols for Matrix 1: ");
    scanf("%d %d", &rows, &cols);
    printf("Enter  Cols for Matrix 2: ");
    scanf("%d", &colsB);

    if(rows > SIZE || cols > SIZE || colsB > SIZE) {
        printf("Size exceeds limit of %d\n", SIZE);
        return 1;
    }

    printf("\nEnter Elements for Matrix 1:\n");
    for(mi=0; mi<rows; mi++)
        for(mj=0; mj<cols; mj++)
            scanf("%d", &Matrix1[mi][mj]);

    printf("\nEnter Elements for Matrix 2 (Rows: %d, Cols: %d):\n", cols, colsB);
    for(mi=0; mi<cols; mi++)
        for(mj=0; mj<colsB; mj++)
            scanf("%d", &Matrix2[mi][mj]);

    // 2. Execute Threads
    pthread_create(&addThread, NULL, addMatrix, NULL);
    pthread_create(&mulThread, NULL, mulMatrix, NULL);

    pthread_join(addThread, NULL);
    pthread_join(mulThread, NULL);

    // 3. Display Resultant Matrices
    printf("\n--- Result of Addition ---\n");
    for(mi=0; mi<rows; mi++) {
        for(mj=0; mj<cols; mj++) printf("%d ", aMatrix[mi][mj]);
        printf("\n");
    }

    printf("\n--- Result of Multiplication ---\n");
    for(mi=0; mi<rows; mi++) {
        for(mj=0; mj<colsB; mj++) printf("%d ", mMatrix[mi][mj]);
        printf("\n");
    }

    // 4. Display ONLY Computation Time

    printf("Addition Time:       %.8f seconds\n", add_exec_time);
    printf("Multiplication Time: %.8f seconds\n", mul_exec_time);

    return 0;
}
